//@version=6
// =============================================================================================
//  INSTITUTIONAL SMC DECISION ENGINE — 15M CRYPTO FUTURES — V2.0
//  ---------------------------------------------------------------------------------------------
//  AUDIT SUMMARY (V1.0 -> V2.0), preserving every working V1 module while upgrading the trigger
//  architecture and classification depth per institutional review:
//
//   1. DECISION ENGINE  — V1 fired on a single weighted score. V2 replaces this with a
//      SEQUENTIAL STATE MACHINE (Bias -> Sweep -> CHoCH -> BOS -> Return-to-Zone -> OB Valid ->
//      FVG Valid -> Momentum -> Volume -> Risk -> Final Decision). The weighted score from V1 is
//      KEPT, but demoted to a confidence/grading layer applied only after the state machine
//      completes — exactly the "confidence layer, not primary trigger" requirement.
//   2. STRUCTURE — added Weak/Strong High/Low classification (protected levels = Strong;
//      just-broken levels = Weak/liquidity-taken) on top of existing internal/external BOS/CHoCH.
//   3. ORDER BLOCKS — added Nested (higher-quality, contained inside an older fresh OB) and
//      Invalidated (deep close-through, not just touched) classification; mitigated OBs that
//      invalidate deeply are flagged as Breaker candidates.
//   4. FVG — added a 4-state lifecycle per gap: FRESH -> PARTIAL -> FILLED/INVALID -> INVERTED
//      (a fully-filled FVG that then gets a strong opposite-direction rejection candle).
//   5. LIQUIDITY — split Internal vs External equal-high/low pools (compares swings of the same
//      class to each other, not across classes as V1 did), added Liquidity Void and Liquidity
//      Consumption (post-sweep displacement confirms the sweep was "real").
//   6. ADAPTIVE SCORING — fixed weights are gone. Market regime (Trending / Ranging / High-Vol /
//      Low-Vol, via ADX + ATR%) multiplies the base weights every bar before scoring.
//   7. VOLUME — added Effort-vs-Result (volume vs range achieved) and Absorption (high volume,
//      small body = supply/demand being absorbed) plus a simple intrabar buy/sell delta proxy.
//   8. RISK — stop/target multiples now scale with regime, added trade expiration (max holding
//      bars), explicit partial-TP flag, invalidation level, and a live 0-100 Trade Health score.
//   9. MARKET FILTERS — ATR compression, extreme volatility, abnormal candle expansion, weak
//      trend and dead/sideways conditions are computed once and reused as hard risk-stage gates.
//  10. PERFORMANCE — every ta.* / request.security() call happens exactly once per bar and is
//      cached into a plain variable; OB/FVG bookkeeping loops are bounded by maxOB/maxFVG.
//  11. DASHBOARD — rebuilt as a PASS/FAIL audit table per module with a plain-language rejection
//      reason when no signal is active, plus confidence % and a letter grade (A+ to D).
//  12/13. Non-repaint / lookahead: unchanged from V1 (HTF security calls use close[1] +
//      lookahead_off; pivots only resolve after `len` bars; signals gate on barstate.isconfirmed).
// =============================================================================================

biasBullOK = trendBullScore >= 50.0
biasBearOK = trendBearScore >= 50.0

atrValPre = ta.atr(atrLen)

regimeTrending = adxVal >= adxTrendMin

regimeHighVol  = atrPercPre > (atrLowPerc + atrHighPerc) / 2.0
regimeLowVol   = atrPercPre < atrLowPerc * 1.5

regimeLabel = regimeTrending ? "TRENDING" : regimeRanging ? "RANGING" : "TRANSITIONAL"
regimeVolLabel = regimeHighVol ? "HIGH-VOL" : regimeLowVol ? "LOW-VOL" : "NORMAL-VOL"

// #############################################################################################
// MODULE 2 — MARKET STRUCTURE ENGINE (BOS / CHoCH / WEAK-STRONG / PREMIUM-DISCOUNT)
// #############################################################################################
extPH = ta.pivothigh(extLen, extLen)
extPL = ta.pivotlow(extLen, extLen)
intPH = ta.pivothigh(intLen, intLen)
intPL = ta.pivotlow(intLen, intLen)

var float extSwingHigh = na
var float extSwingLow  = na
var float intSwingHigh = na
var float intSwingLow  = na
var float prevExtSwingLow  = na
var float prevExtSwingHigh = na
var float extHighHist1 = na
var float extHighHist2 = na
var float extLowHist1  = na
var float extLowHist2  = na
var float intHighHist1 = na
var float intHighHist2 = na
var float intLowHist1  = na
var float intLowHist2  = na

if not na(extPH)
    prevExtSwingHigh := extSwingHigh
    extSwingHigh := extPH
    extHighHist2 := extHighHist1
    extHighHist1 := extPH
if not na(extPL)
    prevExtSwingLow := extSwingLow
    extSwingLow := extPL
    extLowHist2 := extLowHist1
    extLowHist1 := extPL
if not na(intPH)
    intSwingHigh := intPH
    intHighHist2 := intHighHist1
    intHighHist1 := intPH
if not na(intPL)
    intSwingLow := intPL
    intLowHist2 := intLowHist1
    intLowHist1 := intPL

var int extTrend = 0
var int intTrend = 0
var float protectedLow  = na
var float protectedHigh = na
var float weakLowLevel  = na
var float weakHighLevel = na

var bool extBOSbullSignal = false
var bool extBOSbearSignal = false
var bool intBOSbullSignal = false
var bool intBOSbearSignal = false

extBOSbullSignal := close > extSwingHigh
extBOSbearSignal := close < extSwingLow
intBOSbullSignal := close > intSwingHigh
intBOSbearSignal := close < intSwingLow

extBOSbull = confirmedBar and not na(extSwingHigh) and extBOSbullSignal
extBOSbear = confirmedBar and not na(extSwingLow)  and extBOSbearSignal
intBOSbull = confirmedBar and not na(intSwingHigh) and intBOSbullSignal
intBOSbear = confirmedBar and not na(intSwingLow)  and intBOSbearSignal

extCHoCHbull = extBOSbull and extTrend == -1
extCHoCHbear = extBOSbear and extTrend == 1
intCHoCHbull = intBOSbull and intTrend == -1
intCHoCHbear = intBOSbear and intTrend == 1

if extBOSbull
    extTrend := 1
    protectedLow := prevExtSwingLow
    weakHighLevel := extSwingHigh
if extBOSbear
    extTrend := -1
    protectedHigh := prevExtSwingHigh
    weakLowLevel := extSwingLow

if intBOSbull
    intTrend := 1
if intBOSbear
    intTrend := -1

eqLevel = (nz(extSwingHigh, close) + nz(extSwingLow, close)) / 2.0
inPremium  = confirmedBar and not na(extSwingHigh) and not na(extSwingLow) and close > eqLevel
inDiscount = confirmedBar and not na(extSwingHigh) and not na(extSwingLow) and close < eqLevel

structBullScore = math.min(100.0, (extTrend == 1 ? 45.0 : 0.0) + (intTrend == 1 ? 25.0 : 0.0) +
     (extCHoCHbull ? 15.0 : 0.0) + (intCHoCHbull ? 5.0 : 0.0) + (inDiscount ? 10.0 : 0.0))
structBearScore = math.min(100.0, (extTrend == -1 ? 45.0 : 0.0) + (intTrend == -1 ? 25.0 : 0.0) +
     (extCHoCHbear ? 15.0 : 0.0) + (intCHoCHbear ? 5.0 : 0.0) + (inPremium ? 10.0 : 0.0))

// #############################################################################################
// MODULE 3 — LIQUIDITY ENGINE (SWEEPS / INTERNAL-EXTERNAL POOLS / VOID / CONSUMPTION)
// #############################################################################################
atrVal = atrValPre
eqTol = close * eqTolPerc / 100.0

equalHighExternal = not na(extHighHist1) and not na(extHighHist2) and math.abs(extHighHist1 - extHighHist2) <= eqTol
equalLowExternal  = not na(extLowHist1)  and not na(extLowHist2)  and math.abs(extLowHist1  - extLowHist2)  <= eqTol
equalHighInternal = not na(intHighHist1) and not na(intHighHist2) and math.abs(intHighHist1 - intHighHist2) <= eqTol
equalLowInternal  = not na(intLowHist1)  and not na(intLowHist2)  and math.abs(intLowHist1  - intLowHist2)  <= eqTol

sweepBull = confirmedBar and not na(extSwingLow)  and low  <= extSwingLow  and close > extSwingLow
sweepBear = confirmedBar and not na(extSwingHigh) and high >= extSwingHigh and close < extSwingHigh

var int barsSinceSweepBull = 999
var int barsSinceSweepBear = 999
var float sweepBullRefHigh = na
var float sweepBearRefLow  = na
barsSinceSweepBull := sweepBull ? 0 : barsSinceSweepBull + 1
barsSinceSweepBear := sweepBear ? 0 : barsSinceSweepBear + 1
if sweepBull
    sweepBullRefHigh := high
if sweepBear
    sweepBearRefLow := low

sweepBullRecent = barsSinceSweepBull <= sweepLookback
sweepBearRecent = barsSinceSweepBear <= sweepLookback

liquidityConsumedBull = sweepBullRecent and barsSinceSweepBull <= consumptionWindow and not na(sweepBullRefHigh) and close > sweepBullRefHigh
liquidityConsumedBear = sweepBearRecent and barsSinceSweepBear <= consumptionWindow and not na(sweepBearRefLow)  and close < sweepBearRefLow

rangeSizePre = high - low
liquidityVoid = atrVal > 0 and rangeSizePre > atrVal * voidATRMult

lowerWickPre = math.min(close, open) - low
upperWickPre = high - math.max(close, open)
bodySizePre  = math.abs(close - open)
stopHuntBull = sweepBull and lowerWickPre > bodySizePre * 2.5
stopHuntBear = sweepBear and upperWickPre > bodySizePre * 2.5

liquidityBullScore = math.min(100.0, (sweepBullRecent ? 75.0 : 0.0) + (liquidityConsumedBull ? 25.0 : 0.0) + (equalLowExternal or equalLowInternal ? 15.0 : 0.0))
liquidityBearScore = math.min(100.0, (sweepBearRecent ? 75.0 : 0.0) + (liquidityConsumedBear ? 25.0 : 0.0) + (equalHighExternal or equalHighInternal ? 15.0 : 0.0))

if showSwe and confirmedBar and sweepBull
    label.new(bar_index, low, stopHuntBull ? "HUNT" : "SFP", style=label.style_label_up, color=sweepBullLabelColor, textcolor=color.white, size=size.tiny)
if showSwe and confirmedBar and sweepBear
    label.new(bar_index, high, stopHuntBear ? "HUNT" : "SFP", style=label.style_label_down, color=sweepBearLabelColor, textcolor=color.white, size=size.tiny)

bullCandleGlobal = close > open
bearCandleGlobal = close < open

// #############################################################################################
// MODULE 4 — ORDER BLOCK ENGINE (FRESH / MITIGATED / BREAKER / INVALIDATED / NESTED)
// #############################################################################################
var array<box>   bullOBBoxes  = array.new<box>()
var array<box>   bearOBBoxes  = array.new<box>()
var array<float> bullOBTop    = array.new<float>()
var array<float> bullOBBot    = array.new<float>()
var array<bool>  bullOBFresh  = array.new<bool>()
var array<bool>  bullOBNested = array.new<bool>()
var array<bool>  bullOBBreaker = array.new<bool>()
var array<float> bearOBTop    = array.new<float>()
var array<float> bearOBBot    = array.new<float>()
var array<bool>  bearOBFresh  = array.new<bool>()
var array<bool>  bearOBNested = array.new<bool>()
var array<bool>  bearOBBreaker = array.new<bool>()

if confirmedBar and intBOSbull
    obIdx = -1
    for i = 1 to obSearchBack
        if close[i] < open[i]
            obIdx := i
            break
    if obIdx >= 0
        top = open[obIdx] > close[obIdx] ? open[obIdx] : close[obIdx]
        bot = low[obIdx]
        nested = false
        if array.size(bullOBBoxes) > 0
            size = array.size(bullOBBoxes)
            for i = 0 to size - 1
                if array.get(bullOBFresh, i) and top <= array.get(bullOBTop, i) and bot >= array.get(bullOBBot, i)
                    nested := true
        bx = box.new(bar_index[obIdx], top, bar_index + 20, bot, border_color=colBull, bgcolor=bullOBColor, extend=extend.none)
        array.unshift(bullOBBoxes, bx)
        array.unshift(bullOBTop, top)
        array.unshift(bullOBBot, bot)
        array.unshift(bullOBFresh, true)
        array.unshift(bullOBNested, nested)
        array.unshift(bullOBBreaker, false)
        if array.size(bullOBBoxes) > maxOB
            box.delete(array.pop(bullOBBoxes))
            array.pop(bullOBTop)
            array.pop(bullOBBot)
            array.pop(bullOBFresh)
            array.pop(bullOBNested)
            array.pop(bullOBBreaker)

if confirmedBar and intBOSbear
    obIdx = -1
    for i = 1 to obSearchBack
        if close[i] > open[i]
            obIdx := i
            break
    if obIdx >= 0
        top = high[obIdx]
        bot = open[obIdx] < close[obIdx] ? open[obIdx] : close[obIdx]
        nested = false
        if array.size(bearOBBoxes) > 0
            size = array.size(bearOBBoxes)
            for i = 0 to size - 1
                if array.get(bearOBFresh, i) and top <= array.get(bearOBTop, i) and bot >= array.get(bearOBBot, i)
                    nested := true
        bx = box.new(bar_index[obIdx], top, bar_index + 20, bot, border_color=colBear, bgcolor=bearOBColor, extend=extend.none)
        array.unshift(bearOBBoxes, bx)
        array.unshift(bearOBTop, top)
        array.unshift(bearOBBot, bot)
        array.unshift(bearOBFresh, true)
        array.unshift(bearOBNested, nested)
        array.unshift(bearOBBreaker, false)
        if array.size(bearOBBoxes) > maxOB
            box.delete(array.pop(bearOBBoxes))
            array.pop(bearOBTop)
            array.pop(bearOBBot)
            array.pop(bearOBFresh)
            array.pop(bearOBNested)
            array.pop(bearOBBreaker)

priceInFreshBullOB = false
priceInFreshBearOB = false
priceInNestedBullOB = false
priceInNestedBearOB = false

if confirmedBar and array.size(bullOBBoxes) > 0
    obSize = array.size(bullOBBoxes)
    for i = 0 to obSize - 1
        top = array.get(bullOBTop, i)
        bot = array.get(bullOBBot, i)
        fresh = array.get(bullOBFresh, i)
        boxObj = array.get(bullOBBoxes, i)
        if fresh and close < bot - atrVal * obDeepInvalidATR
            array.set(bullOBFresh, i, false)
            array.set(bullOBBreaker, i, true)
            box.set_bgcolor(boxObj, bullOBInvalidColor)
            box.set_border_color(boxObj, bullOBInvalidBorderColor)
        else if fresh and close < bot
