    <script>
        // Mocking Data base to rigorously calculate against user's strict 7-Point SOP Rules
        const targetPairs = ['EURUSD', 'GBPUSD', 'AUDUSD', 'USDJPY', 'XAUUSD', 'BTCUSD', 'ETHUSD'];

        function generateMockTechnicalData(pair) {
            // Randomizer engineered to yield realistic combinations, pushing some into high confluence or blacklist conditions
            const rsi = Math.floor(Math.random() * (75 - 25 + 1)) + 25;
            const ema200 = Math.random() > 0.4; // true if price above EMA200
            const goldenCross = Math.random() > 0.4; 
            const aboveIchimoku = Math.random() > 0.5;
            const macdCross = Math.random() > 0.4;
            const bbPosition = Math.random() > 0.3; // true if close above mid/upper band
            const volumeMultiplier = parseFloat((Math.random() * (2.2 - 0.5) + 0.5).toFixed(2));
            
            // Blacklist factors
            const hasDivergence = Math.random() < 0.12; 
            const atrRatio = parseFloat((Math.random() * (2.5 - 0.5) + 0.5).toFixed(2)); // > 2.0 triggers excessive volatility blacklist
            const upcomingNews = Math.random() < 0.08;

            return { pair, rsi, ema200, goldenCross, aboveIchimoku, macdCross, bbPosition, volumeMultiplier, hasDivergence, atrRatio, upcomingNews };
        }

        function runSOPv2Engine(data) {
            let longScore = 0;
            let shortScore = 0;
            
            let longChecklist = [];
            let shortChecklist = [];

            // --- LONG SCORE CALCULATION ---
            if (data.ema200) { longScore++; longChecklist.push("Weekly/Daily Bullish & > EMA200"); } else { longChecklist.push("FAIL: Di bawah EMA200"); }
            if (data.goldenCross) { longScore++; longChecklist.push("EMA 50 > EMA 200 (Golden Cross Aktif)"); } else { longChecklist.push("FAIL: Death Cross Aktif"); }
            if (data.aboveIchimoku) { longScore++; longChecklist.push("Harga di atas Ichimoku Cloud (4H)"); } else { longChecklist.push("FAIL: Di dalam/bawah Cloud"); }
            if (data.rsi >= 45 && data.rsi <= 65) { longScore++; longChecklist.push(RSI Ideal (${data.rsi}) — Ruang naik sedia ada); } else { longChecklist.push(FAIL: RSI (${data.rsi}) di luar parameter 45-65); }
            if (data.macdCross) { longScore++; longChecklist.push("MACD Bullish Cross + Histogram Positif"); } else { longChecklist.push("FAIL: MACD Bearish Momentum"); }
            if (data.bbPosition) { longScore++; longChecklist.push("Candle close di atas Mid/Upper Bollinger Band"); } else { longChecklist.push("FAIL: Close bawah Mid BB"); }
            if (data.volumeMultiplier >= 1.5) { longScore++; longChecklist.push(Volume Lonjakan Semasa (${data.volumeMultiplier}x > MA20)); } else { longChecklist.push(FAIL: Volume Lemah (${data.volumeMultiplier}x)); }

            // --- SHORT SCORE CALCULATION ---
            if (!data.ema200) { shortScore++; shortChecklist.push("Weekly/Daily Bearish & < EMA200"); } else { shortChecklist.push("FAIL: Di atas EMA200"); }
            if (!data.goldenCross) { shortScore++; shortChecklist.push("EMA 50 < EMA 200 (Death Cross Aktif)"); } else { shortChecklist.push("FAIL: Golden Cross Aktif"); }
            if (!data.aboveIchimoku) { shortScore++; shortChecklist.push("Harga di bawah Ichimoku Cloud (4H)"); } else { shortChecklist.push("FAIL: Di atas/dalam Cloud"); }
            if (data.rsi >= 35 && data.rsi <= 55) { shortScore++; shortChecklist.push(RSI Ideal (${data.rsi}) — Ruang turun sedia ada); } else { shortChecklist.push(FAIL: RSI (${data.rsi}) di luar parameter 35-55); }
            if (!data.macdCross) { shortScore++; shortChecklist.push("MACD Bearish Cross + Histogram Negatif"); } else { shortChecklist.push("FAIL: MACD Bullish Momentum"); }
            if (!data.bbPosition) { shortScore++; shortChecklist.push("Candle close di bawah Mid/Lower Bollinger Band"); } else { shortChecklist.push("FAIL: Close atas Mid BB"); }
            if (data.volumeMultiplier >= 1.5) { shortScore++; shortChecklist.push(Volume Lonjakan Semasa (${data.volumeMultiplier}x > MA20)); } else { shortChecklist.push(FAIL: Volume Lemah (${data.volumeMultiplier}x)); }

            // Evaluate Blacklist rules immediately
            let isBlacklisted = false;
            let blacklistReason = "";
            if (data.hasDivergence) { isBlacklisted = true; blacklistReason = "RSI Counter-Trend Divergence"; }
            if (data.atrRatio > 2.0) { isBlacklisted = true; blacklistReason = Volatiliti Melampau (ATR ${data.atrRatio}x > Purata); }
            if (data.upcomingNews) { isBlacklisted = true; blacklistReason = "Major High-Impact News Hadapan (< 4H)"; }

            let structuralPrice = data.pair.includes('XAU') ? 2450.00 : (data.pair.includes('BTC') ? 92500 : 1.1250);
            let atrValue = data.pair.includes('XAU') ? 14.5 : (data.pair.includes('BTC') ? 1200 : 0.0035);

            return {
                pair: data.pair,
                currentPrice: structuralPrice,
                atr: atrValue,
                longScore,
                shortScore,
                longChecklist,
                shortChecklist,
                isBlacklisted,
                blacklistReason
            };
        }

        function simulateMarketScan() {
            const grid = document.getElementById('signalsGrid');
            const tableBody = document.getElementById('matrixTableBody');
            
            grid.innerHTML = "";
            tableBody.innerHTML = "";

            let activeSignalCount = 0;

            targetPairs.forEach(pair => {
                const rawTech = generateMockTechnicalData(pair);
                const evaluation = runSOPv2Engine(rawTech);

                // Determine final structural output
                let finalState = "NEUTRAL";
                let finalScore = 0;
                let activeChecklist = [];

                if (evaluation.isBlacklisted) {
                    finalState = "BLACKLIST";
                } else if (evaluation.longScore >= 5) {
                    finalState = "LONG";
                    finalScore = evaluation.longScore;
                    activeChecklist = evaluation.longChecklist;
                } else if (evaluation.shortScore >= 5) {
                    finalState = "SHORT";
                    finalScore = evaluation.shortScore;
                    activeChecklist = evaluation.shortChecklist;
                }

                // Append inside monitoring table matrix 
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td style="font-weight:700;">${evaluation.pair}</td>
                    <td>${rawTech.ema200 ? '🟢 Above EMA200' : '🔴 Below EMA200'} · ${rawTech.aboveIchimoku ? 'Cloud Up' : 'Cloud Dn'}</td>
                    <td>${rawTech.rsi}</td>
                    <td>${rawTech.volumeMultiplier}x</td>
                    <td style="font-family:var(--font-mono); font-weight:700;">L:${evaluation.longScore} | S:${evaluation.shortScore}</td>
                    <td><span class="sig-badge" style="background-color:rgba(255,255,255,0.05); color:white;">${finalState === "BLACKLIST" ? '⚠️ BLACKLISTED' : finalState}</span></td>
                `;
                tableBody.appendChild(tr);

                // Inject into prominent Signal Grid Cards only if parameters align to SOP rules
                if (finalState === "LONG" || finalState === "SHORT" || finalState === "BLACKLIST") {
                    activeSignalCount++;
                    
                    let slPrice, tp1, tp2, tp3;
                    if (finalState === "LONG") {
                        slPrice = evaluation.currentPrice - (evaluation.atr * 1.5);
                        let riskPips = evaluation.currentPrice - slPrice;
                        tp1 = evaluation.currentPrice + (riskPips * 1.5);
                        tp2 = evaluation.currentPrice + (riskPips * 2.5);
                        tp3 = evaluation.currentPrice + (riskPips * 4.0);
                    } else if (finalState === "SHORT") {
                        slPrice = evaluation.currentPrice + (evaluation.atr * 1.5);
                        let riskPips = slPrice - evaluation.currentPrice;
                        tp1 = evaluation.currentPrice - (riskPips * 1.5);
                        tp2 = evaluation.currentPrice - (riskPips * 2.5);
                        tp3 = evaluation.currentPrice - (riskPips * 4.0);
                    }

                    const card = document.createElement('div');
                    card.className = sig-card ${finalState};
                    
                    let checklistHTML = activeChecklist.map(item => {
                        const isFail = item.startsWith("FAIL");
                        return `<div class="checklist-item ${isFail ? 'unchecked' : 'checked'}">
                            ${isFail ? '❌' : '✅'} ${item}
                        </div>`;
                    }).join('');

                    if (finalState === "BLACKLIST") {
                        card.innerHTML = `
                            <div class="sig-header">
                                <span class="sig-pair">${evaluation.pair}</span>
                                <span class="sig-badge">⚠️ BLOCKED</span>
                            </div>
                            <div class="score-box" style="color:var(--color-warning);">
                                REASON: ${evaluation.blacklistReason}
                            </div>
                            <p style="font-size:12px; color:var(--text-secondary);">SOP Blocklist protection mechanism dynamically applied to protect trading capital.</p>
                        `;
                    } else {
                        card.innerHTML = `
                            <div class="sig-header">
                                <span class="sig-pair">${evaluation.pair}</span>
                                <span class="sig-badge">${finalState} ENTRY</span>
                            </div>
                            <div class="score-box">
                                Confluence Score: <strong>${finalScore}/7 Matched</strong>
                            </div>
                            <div style="margin-bottom:10px;">
                                ${checklistHTML}
                            </div>
                            <div class="levels-box">
                                <div class="level-item"><span>Entry Price</span><p>${evaluation.currentPrice.toLocaleString()}</p></div>
                                <div class="level-item"><span style="color:var(--color-bearish)">Dynamic SL (1.5x ATR)</span><p style="color:var(--color-bearish)">${slPrice.toLocaleString()}</p></div>
                                <div class="level-item"><span style="color:var(--color-bullish)">TP1 (1:1.5 RR -> SL BE)</span><p>${tp1.toLocaleString()}</p></div>
                                <div class="level-item"><span style="color:var(--color-bullish)">TP2 (1:2.5 RR)</span><p>${tp2.toLocaleString()}</p></div>
                                <div class="level-item" style="grid-column: span 2; margin-top:4px;"><span style="color:var(--color-warning)">TP3 (1:4.0+ Trailing Stop via EMA21)</span><p>${tp3.toLocaleString()}</p></div>
                            </div>
                        `;
                    }
                    grid.appendChild(card);
                }
            });

            if (activeSignalCount === 0) {
                grid.innerHTML = `<div style="grid-column: 1/-1; text-align:center; padding:40px; color:var(--text-secondary);">
                    No system matches found meeting minimum 5/7 Confluence criteria. Standing by...
                </div>`;
            }
        }

        function calculateLot() {
            const balance = parseFloat(document.getElementById('calcBalance').value) || 0;
            const riskPct = parseFloat(document.getElementById('calcRisk').value) || 0;
            const slPips = parseFloat(document.getElementById('calcSL').value) || 1;
            const pipVal = parseFloat(document.getElementById('calcPipVal').value) || 1;

            // Lot Size Formula = (Modal * Risk%) / (SL dalam pips * Nilai pip)
            const cashRisk = balance * (riskPct / 100);
            const lotSize = cashRisk / (slPips * pipVal);

            document.getElementById('calcResultBox').textContent = RECOMMENDED LOT: ${lotSize.toFixed(2)};
        }

        // Initialize display rules on startup
        window.onload = function() {
            simulateMarketScan();
            calculateLot();
        };
    </script>
